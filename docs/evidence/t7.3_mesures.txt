===================================== SCRIPT (t73_mesures.py) =====================================
"""Mesures de la 7.3 : v1 refaite à l'identique, MAE par semaine, baisse par zone, PSI de fin mars.
Lancé depuis la racine du dépôt : uv run python <ce fichier>. N'écrit rien, n'utilise pas MLflow."""
import os, sys
sys.path.insert(0, os.getcwd())
import pandas as pd, yaml
from train import load_params, fit, mae, baseline_mae
from evaluate import weekly_mae
from drift import make_reference, psi

params = load_params()
policy = yaml.safe_load(open("policy.yaml"))
df = pd.read_parquet(params["data"]["features"])
train = df[df["split"] == "train"]
model = fit(params, train)  # même configuration et même graine que la v1 (§5)

print("== 1. v1 refaite sur data/features.parquet (md5 335cc10a28f85363c7bac4a41a0c58d4) ==")
valid = df[df["split"] == "valid"]
print(f"mae_valid {mae(model, valid):.4f} | mae_baseline_valid {baseline_mae(valid):.4f} | mean_y valid {valid['y'].mean():.2f}")
holdout = df[(df["ts"] >= "2020-03-30 00:00") & (df["ts"] <= "2020-03-31 23:00")]
print(f"MAE sur le holdout du gate (30-31 mars, {len(holdout)} lignes) : {mae(model, holdout):.4f}  (decisions.jsonl : mae_production 21.6775)")

print("\n== 2. MAE par semaine, lignes valid et replay (evaluate.weekly_mae) ==")
print(weekly_mae(model, df).to_string(index=False))

print("\n== 3. Prises en charge moyennes par heure et par zone : 17-29 février (valid) contre 23-31 mars ==")
feb = valid.groupby("zone")["y"].mean()
endmar = df[df["ts"] >= "2020-03-23"].groupby("zone")["y"].mean()
z = pd.DataFrame({"fevrier": feb.round(1), "fin_mars": endmar.round(1)})
z["variation_%"] = ((z["fin_mars"] / z["fevrier"] - 1) * 100).round(1)
print(z.to_string())
print(f"zones en baisse : {(z['variation_%'] < 0).sum()} sur {len(z)} ; variation de {z['variation_%'].min()} % à {z['variation_%'].max()} %")
print(f"heures à 0 sur 20 zones, 23-31 mars : {(df[df['ts'] >= '2020-03-23']['y'] == 0).mean():.1%} des lignes")

print("\n== 4. Prises en charge par jour, 20 zones, mars 2020 ==")
mar = df[df["split"] == "replay"]
daily = mar.groupby(mar["ts"].dt.date)["y"].sum().astype(int)
print(daily.to_string())

print("\n== 5. PSI de la dernière fenêtre (480 lignes = 24 h x 20 zones) de mars contre la référence d'entraînement ==")
ref = make_reference(train, policy["drift"]["features"], policy["drift"]["bins"])
last = mar.sort_values(["ts", "zone"]).tail(policy["drift"]["window"])
for f in policy["drift"]["features"]:
    print(f"{f}: PSI {psi(ref[f], last[f].to_numpy()):.4f}")

"""Mesure complémentaire de la 7.3 : répartition par classe de la dernière fenêtre de mars contre la référence."""
import os, sys
sys.path.insert(0, os.getcwd())
import numpy as np, pandas as pd, yaml
from drift import make_reference, psi

policy = yaml.safe_load(open("policy.yaml"))
df = pd.read_parquet("data/features.parquet")
ref = make_reference(df[df["split"] == "train"], policy["drift"]["features"], policy["drift"]["bins"])
mar = df[df["split"] == "replay"].sort_values(["ts", "zone"])
for name, sample in [("dernière fenêtre de mars (31/03)", mar.tail(policy["drift"]["window"])),
                     ("dernière fenêtre de février (29/02)", df[df["split"] == "valid"].sort_values(["ts", "zone"]).tail(policy["drift"]["window"]))]:
    print(f"\n-- {name} --")
    for f in policy["drift"]["features"]:
        e = np.array(ref[f]["edges"]); s = sample[f].to_numpy()
        counts = np.histogram(np.clip(s, e[0], e[-1]), e)[0]
        live = counts / counts.sum()
        print(f"{f}: PSI {psi(ref[f], s):.4f}")
        for i in range(len(e) - 1):
            print(f"  classe [{e[i]:7.1f} ; {e[i+1]:7.1f}]  référence {ref[f]['shares'][i]:6.1%}   fenêtre {live[i]:6.1%}")
